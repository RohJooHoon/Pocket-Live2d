cmake_minimum_required(VERSION 3.22.1)
project(pocket_live2d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PLUGIN_ROOT ${CMAKE_CURRENT_LIST_DIR}/..)
set(SDK ${PLUGIN_ROOT}/cubism)
add_library(pocket_live2d SHARED src/main/cpp/PocketJni.cpp ${PLUGIN_ROOT}/common/PocketCubismRuntime.cpp)
target_include_directories(pocket_live2d PRIVATE ${PLUGIN_ROOT}/common)
target_link_libraries(pocket_live2d android GLESv2 log)
target_link_options(pocket_live2d PRIVATE "-Wl,-z,max-page-size=16384")
if(EXISTS "${SDK}/Core/include/Live2DCubismCore.h")
    if(NOT EXISTS "${SDK}/Core/lib/android/${ANDROID_ABI}/libLive2DCubismCore.a")
        message(FATAL_ERROR "Cubism Core is missing for ${ANDROID_ABI}")
    endif()
    add_library(Live2DCubismCore STATIC IMPORTED)
    set_target_properties(Live2DCubismCore PROPERTIES
        IMPORTED_LOCATION "${SDK}/Core/lib/android/${ANDROID_ABI}/libLive2DCubismCore.a"
        INTERFACE_INCLUDE_DIRECTORIES "${SDK}/Core/include")
    set(FRAMEWORK_SOURCE OpenGL)
    add_subdirectory("${SDK}/Framework" "${CMAKE_CURRENT_BINARY_DIR}/Framework")
    target_compile_definitions(Framework PUBLIC CSM_TARGET_ANDROID_ES2)
    target_link_libraries(Framework Live2DCubismCore)
    target_include_directories(pocket_live2d PRIVATE "${SDK}/include")
    target_compile_definitions(pocket_live2d PRIVATE POCKET_CUBISM_ENABLED=1)
    target_link_libraries(pocket_live2d Framework Live2DCubismCore)
else()
    message(STATUS "Cubism SDK absent: compiling an explicit sdk_unavailable adapter.")
    target_compile_definitions(pocket_live2d PRIVATE POCKET_CUBISM_ENABLED=0)
endif()
